cmake_minimum_required(VERSION 3.24)
project(pi_kanban_native_candidate LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This candidate requires native Windows 11 SDK. Linux is not an isolation validation target.")
endif()
add_library(pi_kanban_launcher STATIC launcher.cpp provision.cpp)
target_compile_features(pi_kanban_launcher PRIVATE cxx_std_20)
target_compile_definitions(pi_kanban_launcher PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(pi_kanban_launcher PRIVATE userenv advapi32 bcrypt onecoreuap)
if(MSVC)
  target_compile_options(pi_kanban_launcher PRIVATE /W4 /WX /permissive- /sdl /guard:cf)
endif()
add_executable(pi_kanban_native_smoke smoke.cpp)
target_compile_features(pi_kanban_native_smoke PRIVATE cxx_std_20)
target_compile_definitions(pi_kanban_native_smoke PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(pi_kanban_native_smoke PRIVATE pi_kanban_launcher userenv advapi32 bcrypt onecoreuap ws2_32)
if(MSVC)
  target_compile_options(pi_kanban_native_smoke PRIVATE /W4 /WX /permissive- /sdl /guard:cf)
endif()
enable_testing()
add_test(NAME native_candidate_smoke COMMAND pi_kanban_native_smoke)
set_tests_properties(native_candidate_smoke PROPERTIES TIMEOUT 60)
add_test(NAME native_registry_read_smoke COMMAND pi_kanban_native_smoke --registry-read)
set_tests_properties(native_registry_read_smoke PROPERTIES TIMEOUT 60)
add_test(NAME native_appcontainer_smoke COMMAND pi_kanban_native_smoke --appcontainer)
set_tests_properties(native_appcontainer_smoke PROPERTIES TIMEOUT 60)
add_executable(pi_kanban_native_helper helper.cpp)
target_compile_features(pi_kanban_native_helper PRIVATE cxx_std_20)
target_compile_definitions(pi_kanban_native_helper PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(pi_kanban_native_helper PRIVATE pi_kanban_launcher userenv advapi32 bcrypt onecoreuap)
if(MSVC)
  target_compile_options(pi_kanban_native_helper PRIVATE /W4 /WX /permissive- /sdl /guard:cf)
endif()
